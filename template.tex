% Numeriamo solo i capitoli (livello section): l'etichetta "Capitolo N" e' generata
% da LaTeX (vedi \titleformat piu' sotto). Le sottosezioni restano senza numero.
% Introduzione, Conclusione e Appendice sono heading non numerati (\section*).
\setcounter{secnumdepth}{1}

% Apertura capitolo: etichetta "CAPITOLO N" in accento su riga a se', titolo sotto.
% I capitoli non numerati (\section*: Introduzione/Conclusione/Appendice) mostrano
% solo il titolo, senza etichetta.
\titleformat{\section}
  [display]
  {\sffamily\color{Dark}\raggedright}
  {\large\bfseries\color{Accent}\MakeUppercase{$if(chapterlabel)$$chapterlabel$$else$Capitolo$endif$ \thesection}}
  {0.2em}
  {\LARGE\bfseries}

  {\thesubsection}
  {0.8em}
  {}

  {\thesubsubsection}
  {0.8em}
  {}

% Blocchi di codice SENZA linguaggio (fence ``` semplice): Pandoc li emette come
% ambiente `verbatim` standard, che NON manda a capo le righe lunghe e sfora il
% margine destro (es. output di --profile in cap. 8 e report di mutation in cap. 11).
% Lo ridefiniamo su fvextra con le stesse impostazioni di Highlighting.
\DefineVerbatimEnvironment{verbatim}{Verbatim}{
  breaklines=true,
  breakanywhere=true,
  fontsize=\footnotesize
}

% Codice inline (\texttt di Pandoc): consenti l'andata a capo per non far sforare
% nel margine token lunghi come "Task::factory()->create()". Meccanismo nativo
% XeTeX (veloce): dopo gli operatori si inserisce un punto di interruzione
% (\allowbreak, senza trattino), attivo SOLO dentro \texttt. I blocchi di codice
% sopra usano fvextra e non sono toccati.
\NewCommandCopy\origtexttt\texttt
\XeTeXcharclass`\.=6 \XeTeXcharclass`\:=6 \XeTeXcharclass`\(=6 \XeTeXcharclass`\)=6
\XeTeXcharclass`\>=6 \XeTeXcharclass`\<=6 \XeTeXcharclass`\/=6 \XeTeXcharclass`\-=6
\XeTeXcharclass`\_=6 \XeTeXcharclass`\;=6 \XeTeXcharclass`\,=6 \XeTeXcharclass`\|=6
\XeTeXinterchartoks 6 0={\allowbreak}
\XeTeXinterchartoks 6 6={\allowbreak}
% Robusta: non espansa nell'indice o nei mark di pagina, solo in composizione.
\DeclareRobustCommand{\texttt}[1]{{\XeTeXinterchartokenstate=1\relax\origtexttt{#1}}}
% Nei segnalibri PDF usa il testo semplice (evita comandi non gestiti da hyperref).
\makeatletter
\pdfstringdefDisableCommands{\def\texttt#1{#1}\def\origtexttt#1{#1}}
\makeatother
